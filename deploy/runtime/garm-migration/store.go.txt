type expected struct {
	Token, Controller, User, Blob, JWT string
	Object                             uint
	Private                            map[string]string
}

func must(e error) {
	if e != nil {
		panic("synthetic operation failed")
	}
}
func require(b bool) {
	if !b {
		panic("synthetic invariant failed")
	}
}
func random() string {
	b := make([]byte, 16)
	_, e := rand.Read(b)
	must(e)
	return hex.EncodeToString(b)
}
func digest(b []byte) string { v := sha256.Sum256(b); return hex.EncodeToString(v[:]) }
func private(path string, dir bool) {
	s, e := os.Lstat(path)
	must(e)
	if dir {
		require(s.IsDir() && s.Mode().Perm() == 0700)
	} else {
		require(s.Mode().IsRegular() && s.Mode().Perm() == 0600)
	}
	st := s.Sys().(*syscall.Stat_t)
	require(st.Uid == uint32(os.Getuid()) && st.Gid == uint32(os.Getgid()))
	if !dir {
		require(st.Nlink == 1)
	}
}
func open(path string) *sql.DB {
	d, e := sql.Open("sqlite3", "file:"+path+"?mode=ro&_query_only=1")
	must(e)
	must(d.Ping())
	return d
}
