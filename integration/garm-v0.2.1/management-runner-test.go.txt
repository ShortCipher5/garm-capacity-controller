// Uses the pinned upstream Runner and SQLite Store, not a reimplementation of
// UpdatePoolParams. No providers, GitHub requests or pool worker loops are run.
func (s *PoolTestSuite) TestCapacityManagementAllowlist() {
 before:=s.Fixtures.Pools[0]
 instance,err:=s.Fixtures.Store.CreateInstance(s.adminCtx,before.ID,params.CreateInstanceParams{Name:"fixture-active-runner",OSType:commonParams.Linux,Status:commonParams.InstanceRunning,RunnerStatus:params.RunnerActive,Generation:before.Generation})
 s.Require().NoError(err)
 var update params.UpdatePoolParams
 s.Require().NoError(json.Unmarshal([]byte(`{"tags":["linux","self-hosted"],"priority":12,"max_runners":8,"min_idle_runners":3,"enabled":false}`),&update))
 after,err:=s.Runner.UpdatePoolByID(s.adminCtx,before.ID,update)
 s.Require().NoError(err)
 s.Require().Equal(before.Generation,after.Generation)
 s.Require().Equal(before.Image,after.Image)
 s.Require().Equal(before.Flavor,after.Flavor)
 s.Require().Equal(before.ExtraSpecs,after.ExtraSpecs)
 s.Require().Equal(before.EnableShell,after.EnableShell)
 s.Require().Equal(uint(8),after.MaxRunners)
 s.Require().Equal(uint(3),after.MinIdleRunners)
 s.Require().Equal(uint(12),after.Priority)
 s.Require().False(after.Enabled)
 s.Require().Len(after.Tags,2)
 current,err:=s.Fixtures.Store.GetInstance(s.adminCtx,instance.Name)
 s.Require().NoError(err)
 s.Require().Equal(instance.Generation,current.Generation)
 s.Require().Equal(commonParams.InstanceRunning,current.Status)
 s.Require().Equal(params.RunnerActive,current.RunnerStatus)
 // The actual upstream validation rejects an inconsistent pair.
 max:=uint(2)
 _,err=s.Runner.UpdatePoolByID(s.adminCtx,before.ID,params.UpdatePoolParams{MaxRunners:&max})
 s.Require().Error(err)
}

// A canceled HTTP/request context is not proof that an already accepted settings
// transaction cannot commit. The pinned Runner/Store do not prevent the transaction here.
func (s *PoolTestSuite) TestCapacityManagementCanceledRequestIsNotFence() {
 before:=s.Fixtures.Pools[0]
 ctx,cancel:=context.WithCancel(s.adminCtx)
 cancel()
 priority:=uint(44)
 after,err:=s.Runner.UpdatePoolByID(ctx,before.ID,params.UpdatePoolParams{Priority:&priority})
 s.Require().NoError(err)
 s.Require().Equal(priority,after.Priority)
 observed,err:=s.Fixtures.Store.GetPoolByID(s.adminCtx,before.ID)
 s.Require().NoError(err)
 s.Require().Equal(priority,observed.Priority)
}

// Actual stock public Runner API: disabled/min0 creation still requires max>0
// and an existing positive bootstrap template. No provider validation/execution
// happens here. The native template-zero mode used by our new foundation needs
// the separately qualified source8 patch and opted provider configuration.
func (s *RepoTestSuite) TestCapacityManagementDisabledPoolCreateBoundary() {
 tpl,err:=s.Fixtures.Store.CreateTemplate(s.Fixtures.AdminContext,params.CreateTemplateParams{Name:"disposable-disabled-template",Data:[]byte("not-executed"),OSType:commonParams.Linux,ForgeType:params.GithubEndpointType,IsSystem:true})
 s.Require().NoError(err)
 spec:=s.Fixtures.CreatePoolParams
 spec.Enabled=false
 spec.MinIdleRunners=0
 spec.MaxRunners=2
 zero:=uint(0)
 spec.TemplateID=&zero
 repoID:=s.Fixtures.StoreRepos["test-repo-1"].ID
 _,err=s.Runner.CreateRepoPool(s.Fixtures.AdminContext,repoID,spec)
 s.Require().Error(err,"stock/unopted native template-zero is unsupported")
 spec.TemplateID=&tpl.ID
 spec.MaxRunners=0
 _,err=s.Runner.CreateRepoPool(s.Fixtures.AdminContext,repoID,spec)
 s.Require().Error(err,"max0 is rejected even when disabled")
 spec.MaxRunners=2
 pool,err:=s.Runner.CreateRepoPool(s.Fixtures.AdminContext,repoID,spec)
 s.Require().NoError(err)
 s.Require().NotEmpty(pool.ID)
 s.Require().False(pool.Enabled)
 s.Require().Zero(pool.MinIdleRunners)
 s.Require().Equal(uint(2),pool.MaxRunners)
 s.Require().Equal(tpl.ID,pool.TemplateID)
 s.Require().Empty(pool.Instances)
 fresh,err:=s.Runner.GetRepoPoolByID(s.Fixtures.AdminContext,repoID,pool.ID)
 s.Require().NoError(err)
 s.Require().Equal(tpl.ID,fresh.TemplateID)
 s.Require().Equal(repoID,fresh.RepoID)
 s.Require().False(fresh.Enabled)
 s.Require().Empty(fresh.Instances)
 s.Fixtures.ProviderMock.AssertNumberOfCalls(s.T(),"ValidatePoolInfo",0)
 repo,err:=s.Fixtures.Store.GetRepositoryByID(s.Fixtures.AdminContext,repoID)
 s.Require().NoError(err)
 s.Require().Len(repo.Pools,1,"failed attempts did not insert extra pools")
}

// Schedule a second official Runner call after the original Runner has
// authorized and validated its request, immediately before the real SQLite
// store runs its transaction. Both callers use the unmodified store.
type capacityManagementInterleavedStore struct {
	dbCommon.Store
	beforeUpdate func()
}

func (s capacityManagementInterleavedStore) UpdateEntityPool(ctx context.Context, entity params.ForgeEntity, poolID string, update params.UpdatePoolParams) (params.Pool, error) {
	s.beforeUpdate()
	return s.Store.UpdateEntityPool(ctx, entity, poolID, update)
}

func (s *PoolTestSuite) TestCapacityManagementMatchingMaximumBeforeOriginalCommits() {
	pool := s.Fixtures.Pools[0]
	desired := pool.MaxRunners + 1
	other := desired + 1
	ctx, cancel := context.WithCancel(s.adminCtx)
	defer cancel()
	original := &Runner{store: capacityManagementInterleavedStore{
		Store: s.Fixtures.Store,
		beforeUpdate: func() {
			cancel()
			// An independently authorized entity-route call supplies the desired
			// maximum while the original generic-route call has yet to write.
			_, err := s.Runner.UpdateOrgPool(s.adminCtx, pool.OrgID, pool.ID, params.UpdatePoolParams{MaxRunners: &desired})
			s.Require().NoError(err)
			observed, err := s.Runner.GetPoolByID(s.adminCtx, pool.ID)
			s.Require().NoError(err)
			s.Require().Equal(desired, observed.MaxRunners)
			s.Require().Equal(pool.Generation, observed.Generation)
			changed, err := s.Runner.UpdateOrgPool(s.adminCtx, pool.OrgID, pool.ID, params.UpdatePoolParams{MaxRunners: &other})
			s.Require().NoError(err)
			s.Require().Equal(other, changed.MaxRunners)
		},
	}}
	_, err := original.UpdatePoolByID(ctx, pool.ID, params.UpdatePoolParams{MaxRunners: &desired})
	s.Require().NoError(err)
	observed, err := s.Runner.GetPoolByID(s.adminCtx, pool.ID)
	s.Require().NoError(err)
	s.Require().Equal(desired, observed.MaxRunners)
	s.Require().Equal(pool.Generation, observed.Generation)
}
